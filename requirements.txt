# ═══════════════════════════════════════════════════════════════════════════════
#  ShadeMusicBot — requirements.txt
#  Phase 2: Stable Music Engine Migration
#
#  Key decisions:
#   • pyrofork replaces pyrogram — same import namespace, actively maintained,
#     tracks current Telegram API layer, Python >=3.10 compatible.
#   • TgCrypto-pyrofork is pyrofork's crypto companion (same import as TgCrypto).
#   • pytgcalls 0.9.x is the current stable PyPI release under the OLD package
#     name "pytgcalls" (NOT "py-tgcalls" which was a 2.x era rename experiment).
#     As of 2025-2026, the project reverted to "pytgcalls" on PyPI with a
#     completely rewritten API built on ntgcalls.
#   • ntgcalls is the native C++ backend; pulled as a transitive dep of pytgcalls
#     but pinned explicitly to guarantee a wheel-only install on Render.
#   • yt-dlp is unpinned above a floor — always runs latest to track YouTube API.
# ═══════════════════════════════════════════════════════════════════════════════

# ─── Telegram MTProto ──────────────────────────────────────────────────────────
# pyrofork: actively maintained pyrogram fork, same "pyrogram" import namespace.
# TgCrypto-pyrofork: the matching C-extension crypto lib (same "TgCrypto" import).
pyrofork>=2.3.0
TgCrypto-pyrofork>=1.2.5

# ─── Voice Chat Streaming ─────────────────────────────────────────────────────
# pytgcalls: the canonical package name (the "py-tgcalls" alias is discontinued).
# ntgcalls provides pre-built Linux x86_64 wheels — no compilation on Render.
#
# Version strategy:
#   >=0.9.0 ensures the stable ntgcalls-backed API surface:
#     pytgcalls.PyTgCalls
#     pytgcalls.types.MediaStream, AudioQuality, AudioParameters
#     pytgcalls.types.Update, ChatUpdate, StreamAudioEnded
#     pytgcalls.filters (chat_update)
#     client.play(), client.change_stream(), client.leave_call()
#     client.pause_stream(), client.resume_stream()
#
#   <1.0.0 keeps us on a stable minor series; 1.x will be a new major with
#   breaking changes and we will pin forward explicitly when it releases.
pytgcalls>=0.9.0,<1.0.0
ntgcalls>=1.3.0,<2.0.0

# ─── YouTube / Audio extraction ────────────────────────────────────────────────
# Never pin yt-dlp — YouTube changes weekly; stale yt-dlp = silent failures.
# The floor ensures we have modern extractor_args / player_client support.
yt-dlp>=2025.1.0
