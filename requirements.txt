# ─── Telegram MTProto ─────────────────────────────────────────────────────────
# IMPORTANT: do NOT use the original "pyrogram" package here.
#
# The original pyrogram project was abandoned at 2.0.106 and is missing modern
# Telegram MTProto error classes (e.g. GroupCallForbidden) that py-tgcalls
# needs internally at startup. Installing pyrogram==2.0.106 causes:
#   ImportError: cannot import name 'GroupCallForbidden' from pyrogram.errors
#
# pyrofork is the actively maintained fork of pyrogram that:
#   • tracks the current Telegram API layer (all modern error classes included)
#   • installs under the SAME "pyrogram" import namespace — zero code changes
#   • requires Python >=3.10 (we target 3.12, so this is satisfied)
#
# TgCrypto-pyrofork is the corresponding C-extension crypto library for
# pyrofork (replaces TgCrypto, same import name, same API).
pyrofork>=2.3.0
TgCrypto-pyrofork>=1.2.5

# ─── Voice Chat Streaming ─────────────────────────────────────────────────────

#                                    broken tgcalls C++ extension)
# Import name       : pytgcalls    (unchanged — from pytgcalls import PyTgCalls)
# Backend           : ntgcalls — pre-built wheels for Linux x86_64, no compilation
py-tgcalls>=2.0.0

# ntgcalls is a transitive dep of py-tgcalls but pinned explicitly so pip
# always resolves to a wheel-only release (avoids source builds on Render).
ntgcalls>=1.3.0
